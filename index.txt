/* ===================== AVL ENGINE (pure functions) ===================== */
const T=(key,left=null,right=null)=>({key,left,right});
const H=n=>n?1+Math.max(H(n.left),H(n.right)):0;
const BF=n=>n?H(n.left)-H(n.right):0;
const sg=n=>n>0?'+'+n:String(n);
const count=t=>t?1+count(t.left)+count(t.right):0;
const inorder=t=>t?[...inorder(t.left),t.key,...inorder(t.right)]:[];
const treeStr=t=>!t?'∅':(t.left||t.right)?`${t.key}(${treeStr(t.left)}, ${treeStr(t.right)})`:String(t.key);
const rotR=y=>{const x=y.left;return T(x.key,x.left,T(y.key,x.right,y.right))};
const rotL=x=>{const y=x.right;return T(y.key,T(x.key,x.left,y.left),y.right)};
const replace=(t,k,fn)=>!t?t:t.key===k?fn(t):k<t.key?T(t.key,replace(t.left,k,fn),t.right):T(t.key,t.left,replace(t.right,k,fn));
const find=(t,k)=>!t?null:t.key===k?t:find(k<t.key?t.left:t.right,k);
const minNode=t=>t.left?minNode(t.left):t;
const bstIns=(t,k)=>!t?T(k):k<t.key?T(t.key,bstIns(t.left,k),t.right):k>t.key?T(t.key,t.left,bstIns(t.right,k)):t;
const bstDel=(t,k)=>{if(!t)return t;if(k<t.key)return T(t.key,bstDel(t.left,k),t.right);if(k>t.key)return T(t.key,t.left,bstDel(t.right,k));
  if(!t.left)return t.right;if(!t.right)return t.left;const s=minNode(t.right);return T(s.key,t.left,bstDel(t.right,s.key))};
const applyRot=(t,r)=>replace(t,r[1],r[0]==='R'?rotR:rotL);
const RN={R:'Right rotation',L:'Left rotation'};
const FIX={LL:'Single right rotation at the unbalanced node.',RR:'Single left rotation at the unbalanced node.',LR:'Left rotation at the left child, then right rotation at the node.',RL:'Right rotation at the right child, then left rotation at the node.'};

function diagNode(n){
  const bf=BF(n);if(Math.abs(bf)<2)return null;
  if(bf>0){const c=n.left,cb=BF(c),kase=cb>=0?'LL':'LR';
    return {key:n.key,bf,child:c.key,cb,kase,rots:kase==='LL'?[['R',n.key]]:[['L',c.key],['R',n.key]]}}
  const c=n.right,cb=BF(c),kase=cb<=0?'RR':'RL';
  return {key:n.key,bf,child:c.key,cb,kase,rots:kase==='RR'?[['L',n.key]]:[['R',c.key],['L',n.key]]};
}
const diagnose=t=>!t?null:(diagnose(t.left)||diagnose(t.right)||diagNode(t));
const bal=n=>{const d=diagNode(n);if(!d)return n;let t=n;for(const r of d.rots)t=applyRot(t,r);return t};
const avlIns=(t,k)=>!t?T(k):k<t.key?bal(T(t.key,avlIns(t.left,k),t.right)):k>t.key?bal(T(t.key,t.left,avlIns(t.right,k))):t;

function walkUp(tree,anc,key,S,mode){
  let rotated=false;
  for(const p of anc){
    const n=find(tree,p);if(!n)continue;
    const l=H(n.left),r=H(n.right),bf=l-r,bad=Math.abs(bf)>1;
    const base=mode==='insert'?{[key]:'new'}:{};
    S.push({kind:'check',tree,hl:{...base,[p]:bad?'bad':'chk'},stage:3,bfText:`BF(${p}) = ${l} − ${r} = ${sg(bf)}`,title:`Check node ${p}`,
      text:`Height(${p}) = ${H(n)}. Left height = ${l}, right height = ${r}, so BF = ${l} − ${r} = ${sg(bf)}. `+(bad?'That is outside −1…+1 → unbalanced!':'Within −1…+1 → this node is fine.')});

function useTween(target,sig){

    <div className="row"><button className="btn pri big" onClick={()=>begin(MCQS.map(m=>m.id),true)}>START CHALLENGE →</button></div></div>;
  if(phase==='result')return <Result answers={answers} onRetryWrong={()=>begin(wrongIds,false)} onRetryAll={()=>begin(MCQS.map(m=>m.id),true)}/>;
  const q=MCQS.find(m=>m.id===order[pos]);
  return <QuestionView key={q.id+'-'+pos+'-'+retry} q={q} pos={pos} len={order.length} retry={retry} onAnswer={onAnswer} onNext={onNext}/>;
}

function Presenter({stage,onClose}){
  const [s,setS]=useState(0);useEffect(()=>{const t=setInterval(()=>setS(x=>x+1),1000);return()=>clearInterval(t)},[]);
  const n=CONTENT.notes[Math.max(0,stage)];
  return <div className="presenter glass"><div style={{display:'flex',justifyContent:'space-between',alignItems:'center'}}><b>👨‍🏫 Presenter · {pad(Math.floor(s/60))}:{pad(s%60)}</b><button className="btn" onClick={onClose}>✕</button></div>
    <p><b style={{color:'var(--ac)'}}>SAY</b><br/>{n.say}</p><p style={{marginBottom:0}}><b style={{color:'var(--ok)'}}>DO</b><br/>{n.do}</p></div>;
}
function App(){
  const [stage,setStage]=useState(-1);const [teach,setTeach]=useState(false);const [pres,setPres]=useState(false);
  const [focus,setFocus]=useState(null);const [term,setTerm]=useState(null);const [theme,setTheme]=useState('dark');
  const handlers=useRef([]);const stageRef=useRef(-1);stageRef.current=stage;const N=CONTENT.stages.length;
  const reg=useCallback(h=>{handlers.current.push(h);return()=>{handlers.current=handlers.current.filter(x=>x!==h)}},[]);
  const go=useCallback(i=>{setStage(Math.max(-1,Math.min(N-1,i)));setFocus(null);setTerm(null);window.scrollTo({top:0,behavior:'smooth'})},[]);
  useEffect(()=>{document.documentElement.dataset.theme=theme},[theme]);
  useEffect(()=>{document.documentElement.classList.toggle('teach',teach)},[teach]);
  useEffect(()=>{
    const onKey=e=>{
      const tg=e.target.tagName;if(['INPUT','TEXTAREA','SELECT'].includes(tg)||e.ctrlKey||e.metaKey||e.altKey)return;
      if(e.key==='ArrowRight'){e.preventDefault();go(stageRef.current+1)}
      else if(e.key==='ArrowLeft'){e.preventDefault();go(stageRef.current-1)}
      else if(e.key===' '){e.preventDefault();if(document.activeElement&&document.activeElement.blur)document.activeElement.blur();for(const h of [...handlers.current])if(h())return;go(stageRef.current+1)}
      else if(e.key==='Escape'){setFocus(null);setTerm(null)}
      else if(e.key==='t'||e.key==='T')setTeach(x=>!x);
      else if(e.key==='p'||e.key==='P')setPres(x=>!x);
    };
    addEventListener('keydown',onKey);return()=>removeEventListener('keydown',onKey);
  },[go]);
  const pages=[
    <><WhyBinary/><Problem/></>,
    <BstToAvl/>,
    <><BalanceFactor/><WhyRotate/></>,
    <><Rotations/><Decision/></>,
    <CodeSections/>,
    <Sim/>,
    <Playground/>,
    <Predict/>,
    <><Mistakes/><RealWorld/><Complexity/><Recap/><MindMap/></>,
    <Block id="chal" icon="🎯" kicker="Final test" title="CDAC-level AVL Tree MCQ Challenge"><Challenge/></Block>
  ];
  const glos=CONTENT.glossary;
  return <Ctx.Provider value={{teach,focus,setFocus,setTerm,reg}}>
    {stage>=0&&<div className="pbar" style={{width:`${((stage+1)/N)*100}%`}}/>}
    <header className="top glass">
      <div className="brand hide-teach">{CONTENT.icon} {CONTENT.topic}</div>
      <nav className="tabs hide-teach">{CONTENT.stages.map((s,i)=><button key={s} className={'tab '+(i===stage?'on':i<stage?'done':'')} onClick={()=>go(i)}><b>{pad(i+1)}</b><span className="lbl"> {s}</span></button>)}</nav>
      <div className="ctl">
        <button className={'btn '+(teach?'on':'')} onClick={()=>setTeach(!teach)} title="Teaching mode (T)">🎤<span className="lbl"> TEACHING MODE</span></button>
        <button className={'btn '+(pres?'on':'')} onClick={()=>setPres(!pres)} title="Presenter notes (P)">👨‍🏫<span className="lbl"> Presenter</span></button>
        <button className="btn hide-teach" onClick={()=>setTheme(theme==='dark'?'light':'dark')} title="Theme">{theme==='dark'?'☀️':'🌙'}</button>
      </div>
    </header>
    {stage<0?<Hero onStart={()=>go(0)}/>
      :<main className="wrap stage" key={stage}>
        {teach&&<p className="mut" style={{fontSize:'.8rem',marginTop:0}}>🎤 Teaching mode — click a block to spotlight it · Esc to clear</p>}
        {pages[stage]}</main>}
    {stage>=0&&<div className="bar glass">
      <button className="btn" disabled={stage<=0} onClick={()=>go(stage-1)}>← Previous</button>
      <div className="dots">{CONTENT.stages.map((s,i)=><button key={s} aria-label={s} className={'dot '+(i===stage?'on':i<stage?'done':'')} onClick={()=>go(i)}/>)}</div>
      <span className="where">Section {stage+1} of {N} · {CONTENT.stages[stage]}</span>
      <button className="btn pri" disabled={stage>=N-1} onClick={()=>go(stage+1)}>Next →</button>
    </div>}
    {term&&<div className="pop glass"><button className="x" onClick={()=>setTerm(null)}>✕</button><b>💡 {glos[term].t}</b><p>{glos[term].d}</p></div>}
    {pres&&<Presenter stage={stage} onClose={()=>setPres(false)}/>}
  </Ctx.Provider>;
}
ReactDOM.createRoot(document.getElementById('root')).render(<App/>);
</script>
